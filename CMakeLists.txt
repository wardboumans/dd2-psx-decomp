cmake_minimum_required(VERSION 3.24)
project(dd2 C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# SDL3: use an installed copy when there is one, otherwise build the pinned release from source.
option(DD2_VENDORED_SDL "Always build SDL3 from source" OFF)
if(NOT DD2_VENDORED_SDL)
    find_package(SDL3 3.4 CONFIG QUIET)
endif()
if(NOT SDL3_FOUND)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3
        URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.tar.gz
        URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3
        DOWNLOAD_EXTRACT_TIMESTAMP ON)
    FetchContent_MakeAvailable(SDL3)
endif()

function(dd2_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-parameter)
    endif()
endfunction()

# Everything but the entry point, so tests can link the game and library code.
file(GLOB_RECURSE DD2_SOURCES CONFIGURE_DEPENDS src/*.c)
list(REMOVE_ITEM DD2_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src/main.c)
add_library(dd2_core STATIC ${DD2_SOURCES})
target_include_directories(dd2_core PUBLIC src)
target_link_libraries(dd2_core PUBLIC SDL3::SDL3)
if(NOT MSVC)
    target_link_libraries(dd2_core PUBLIC m)
endif()
dd2_warnings(dd2_core)

add_executable(dd2 src/main.c)
target_link_libraries(dd2 PRIVATE dd2_core)
dd2_warnings(dd2)
if(MSVC)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT dd2)
endif()

enable_testing()
add_executable(gpu_test tests/gpu_test.c)
target_link_libraries(gpu_test PRIVATE dd2_core)
dd2_warnings(gpu_test)
add_test(NAME gpu COMMAND gpu_test ${CMAKE_CURRENT_BINARY_DIR}/gpu_test.bmp)
add_executable(gte_test tests/gte_test.c)
target_link_libraries(gte_test PRIVATE dd2_core)
dd2_warnings(gte_test)
add_test(NAME gte COMMAND gte_test)
add_executable(spu_test tests/spu_test.c)
target_link_libraries(spu_test PRIVATE dd2_core)
dd2_warnings(spu_test)
add_test(NAME spu COMMAND spu_test)
add_executable(card_test tests/card_test.c)
target_link_libraries(card_test PRIVATE dd2_core)
dd2_warnings(card_test)
add_test(NAME card COMMAND card_test)
add_executable(press_test tests/press_test.c)
target_link_libraries(press_test PRIVATE dd2_core)
dd2_warnings(press_test)
add_test(NAME press COMMAND press_test)
